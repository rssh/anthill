==================================
Proposal 062 sort goal — a parenthesised goal beside a requirement and an op clause
==================================

namespace m
  enum Text
    sort Trust = ?
    requires is_entity_of(Trust, TrustLevel)
    requires Eq[T = Trust]
    requires anthill.reflect.typing.is_entity_of(Trust, TrustLevel)
    entity text(raw: String)
    operation pick(a: Text[Trust]) -> Unit
      requires neq(a, a)
  end
end
